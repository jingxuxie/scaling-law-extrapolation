\section{Exact Spectral Scaling Curves}

Let $X_j$ be square-integrable, uncorrelated coordinates with $\E X_j=0$ and $\E X_j^2=\lambda_j$. Consider
\begin{equation}
 Y=\sum_{j\ge1}\theta_jX_j+\varepsilon,
 \qquad \E\varepsilon^2=E,
 \label{eq:linreg}
\end{equation}
where $\varepsilon$ is uncorrelated with the covariates and $\sum_j\lambda_j\theta_j^2<\infty$. The population-optimal predictor using the first $M$ coordinates has risk
\begin{equation}
 \Risk(M)=E+\sum_{j>M}q_j,
 \qquad q_j:=\lambda_j\theta_j^2.
 \label{eq:spectral-tail}
\end{equation}
For $\alpha>0$, define
\begin{equation}
 \phi_\alpha(M):=\sum_{j>M}j^{-(1+\alpha)}
 =\zeta(1+\alpha,M+1).
 \label{eq:basis}
\end{equation}

\begin{proposition}[Spectral realizability]
\label{prop:realizability}
For $\alpha_1,\ldots,\alpha_L>0$ and $w_1,\ldots,w_L\ge0$, choosing
$q_j=\sum_{\ell=1}^Lw_\ell j^{-(1+\alpha_\ell)}$ gives the exact risk
\begin{equation}
 \Risk(M)=E+\sum_{\ell=1}^Lw_\ell\phi_{\alpha_\ell}(M),
 \label{eq:mixture}
\end{equation}
with $\phi_\alpha(M)=\alpha^{-1}M^{-\alpha}+O(M^{-1-\alpha})$.
More generally, every finite positive measure $\mu$ on a compact exponent interval induces
\begin{equation}
 \Risk_{E,\mu}(M)=E+\int\phi_\alpha(M)\,d\mu(\alpha)
 \label{eq:continuous-mixture}
\end{equation}
by setting $q_j=\int j^{-1-\alpha}\,d\mu(\alpha)$.
\end{proposition}

\begin{proposition}[The slowest spectral population controls the exponent]
\label{prop:slowest-support}
Let $\mu$ be a finite nonzero positive measure supported on a compact interval in $(0,\infty)$, and let
$\alpha_0=\inf\operatorname{supp}(\mu)$. Then the exponent functional in Eq.~\eqref{eq:exponent-functional} exists for Eq.~\eqref{eq:continuous-mixture} and
\begin{equation}
 a(\Risk_{E,\mu})=\alpha_0.
 \label{eq:slowest-support}
\end{equation}
This remains true when $\mu$ has no atom at $\alpha_0$; slowly varying factors may change, but the log--log exponent does not.
\end{proposition}

\begin{corollary}[Exact powers]
\label{cor:exact-power}
For $A,\alpha>0$, the predictive energies
$q_j=A\{j^{-\alpha}-(j+1)^{-\alpha}\}$ yield
\begin{equation}
 \Risk(M)=E+A(M+1)^{-\alpha}
 \label{eq:exact-power}
\end{equation}
for every integer $M\ge0$.
\end{corollary}

A single power therefore represents one spectral population, while a crossover represents several. Proposition~\ref{prop:slowest-support} explains why an arbitrarily weak slow component is qualitatively important: it is not merely another finite-scale correction, but the population that determines the eventual exponent. Positivity is structural because mixture weights are predictive energies, not an arbitrary regularizer.
