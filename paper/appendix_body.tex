\section{Proof of Spectral Realizability}

\begin{proof}[Proof of Proposition~\ref{prop:realizability}]
The residual of the population-optimal $M$-coordinate predictor is
$Y-\sum_{j\le M}\theta_jX_j=\sum_{j>M}\theta_jX_j+\varepsilon$.
Uncorrelatedness gives
\[
\E\left(Y-\sum_{j\le M}\theta_jX_j\right)^2
=E+\sum_{j>M}\lambda_j\theta_j^2.
\]
Substituting $q_j=\sum_\ell w_\ell j^{-(1+\alpha_\ell)}$ and exchanging the finite $\ell$-sum with the convergent $j$-sum yields Eq.~\eqref{eq:mixture}. Finally, the integral test or one step of Euler--Maclaurin summation gives
\[
\sum_{j>M}j^{-(1+\alpha)}
=\frac{M^{-\alpha}}{\alpha}+O(M^{-1-\alpha}).
\]
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:exact-power}]
The proposed predictive energies are nonnegative because $j\mapsto j^{-\alpha}$ decreases. They are summable, and their tail telescopes:
\[
 \sum_{j>M}A\{j^{-\alpha}-(j+1)^{-\alpha}\}
 =A(M+1)^{-\alpha}.
\]
Substitution into Eq.~\eqref{eq:spectral-tail} proves the claim. Existence of an underlying regression problem is explicit: take independent standard-normal coordinates, set $\lambda_j=1$ and $\theta_j=\sqrt{q_j}$, and take independent noise with variance $E$. Since $\sum_jq_j<\infty$, the response series converges in $L^2$.
\end{proof}

\section{Proof of the Hidden-Crossover Theorem}

We use the elementary tail bounds
\begin{equation}
 \frac{(M+1)^{-\gamma}}{\gamma}
 \le \phi_\gamma(M)
 \le \frac{M^{-\gamma}}{\gamma},
 \qquad \gamma>0.
 \label{eq:integral-bounds}
\end{equation}
They follow by comparing the decreasing function $x^{-1-\gamma}$ with its left and right integrals.

\begin{proof}[Proof of Theorem~\ref{thm:hidden}]
Because $\phi_\alpha(M)$ decreases in $M$,
\[
0\le\Risk_1(M)-\Risk_0(M)=B\phi_\alpha(M)
\le B\phi_\alpha(M_{\min})=\eta
\]
throughout the pilot interval. The Gaussian laws share covariance $\sigma^2I_m$, hence
\[
\KL(P_0\|P_1)
=\frac{1}{2\sigma^2}\sum_{i=1}^m
\bigl(B\phi_\alpha(M_i)\bigr)^2
\le\frac{m\eta^2}{2\sigma^2}.
\]
For $\eta=\sigma/(2\sqrt m)$ this is at most $1/8$, and Pinsker's inequality gives $\TV(P_0,P_1)\le1/4$.

By Proposition~\ref{prop:realizability} and Eq.~\eqref{eq:exponent-functional},
$\Risk_0(M)-E\sim(A/\beta)M^{-\beta}$, whereas
$\Risk_1(M)-E\sim(B/\alpha)M^{-\alpha}$ because $\alpha<\beta$. Their exponent functionals are therefore separated by $\beta-\alpha$. The standard two-point reduction says that for any estimator of a scalar functional separated by $\Delta$,
\[
\max_k\E_k|\widehat\psi-\psi_k|
\ge\frac{\Delta}{4}(1-\TV(P_0,P_1)).
\]
Substituting $\Delta=\beta-\alpha$ and $\TV\le1/4$ gives Eq.~\eqref{eq:exponent-lower}.

For the relative-risk statement, Eq.~\eqref{eq:integral-bounds} yields
\[
\frac{B\phi_\alpha(M)}{A\phi_\beta(M)}
\ge\frac{B\beta}{A\alpha}\frac{M^\beta}{(M+1)^\alpha},
\]
which proves Eq.~\eqref{eq:relative-divergence}.

If $\Risk_1(M)-E\le\tau$, then $B\phi_\alpha(M)\le\tau$ and the lower bound in Eq.~\eqref{eq:integral-bounds} implies
$M+1\ge(B/(\alpha\tau))^{1/\alpha}$. Conversely, choosing
$M\ge(A/(\beta\tau))^{1/\beta}$ makes
$A\phi_\beta(M)\le\tau$. This proves Eq.~\eqref{eq:compute-target} and the divergence of the ratio as $\tau\downarrow0$.
\end{proof}

\section{Proof of the Floor--Exponent Bound}

\begin{lemma}
\label{lem:H}
Let $H(z)=1-(1+z)e^{-z}$. Then $H(z)\ge0$ for all $z\in\R$,
\begin{equation}
 H(z)\le\frac{z^2}{2}e^{|z|}
 \label{eq:H-upper}
\end{equation}
for all $z$, and
\begin{equation}
 H(z)\ge\frac{z^2}{2}e^{-z}
 \label{eq:H-lower}
\end{equation}
for $z\ge0$.
\end{lemma}

\begin{proof}
Since $H(0)=0$ and $H'(z)=ze^{-z}$, zero is the global minimum. Also
$H(z)=\int_0^z ue^{-u}\,du$. Taking absolute values and bounding
$e^{-u}\le e^{|z|}$ along the integration segment gives Eq.~\eqref{eq:H-upper}. For $z\ge0$, use $e^{-u}\ge e^{-z}$ on $[0,z]$ to get Eq.~\eqref{eq:H-lower}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:matched}]
For $u\ge\alpha$, define
\[
 g_u(t)=E+S\left(1-\frac{\alpha}{u}\right)
 +\frac{\alpha S}{u}e^{-ut}.
\]
Then $g_\alpha=f_0$, $g_\beta=f_1$, and
\[
\frac{\partial g_u(t)}{\partial u}
=\frac{\alpha S}{u^2}\left[1-(1+ut)e^{-ut}\right]
=\frac{\alpha S}{u^2}H(ut).
\]
Integrating from $\alpha$ to $\beta$ proves Eq.~\eqref{eq:gap-integral} and nonnegativity. For $|t|\le T$, Lemma~\ref{lem:H} gives
\[
 f_1(t)-f_0(t)
\le\int_\alpha^\beta
 \frac{\alpha S}{u^2}\frac{u^2t^2}{2}e^{|ut|}\,du
\le\frac{\alpha S\delta}{2}e^{\bar\alpha T}T^2.
\]
For $t_\star\ge0$, the lower bound gives
\[
 f_1(t_\star)-f_0(t_\star)
\ge\int_\alpha^\beta
 \frac{\alpha S}{u^2}\frac{u^2t_\star^2}{2}e^{-ut_\star}\,du
\ge\frac{\alpha S\delta}{2}e^{-\bar\alpha t_\star}t_\star^2.
\]

Let $d_i=f_1(t_i)-f_0(t_i)$. Under Eq.~\eqref{eq:delta-choice}, the pilot upper bound implies $|d_i|\le\sigma/(2\sqrt m)$. Therefore
\[
\KL(P_0\|P_1)=\frac{1}{2\sigma^2}\sum_i d_i^2\le\frac18,
\]
and $\TV(P_0,P_1)\le1/4$. Applying the two-point inequality to the target functional, whose separation is at least Eq.~\eqref{eq:target-gap}, yields Eq.~\eqref{eq:target-lower}. Applying it to the exponent functional gives $3\delta/16$.

Finally, set $\delta=\delta_\star$ as defined after Theorem~\ref{thm:matched} and substitute it into Eq.~\eqref{eq:target-lower}. If $\bar\alpha T\le1$ and $\bar\alpha t_\star\le1$, then
$e^{-\bar\alpha(T+t_\star)}\ge e^{-2}$, yielding Eq.~\eqref{eq:horizon-ratio}.
\end{proof}

\section{Proof of Certificate Coverage}

\begin{proof}[Proof of Theorem~\ref{thm:certificate}]
For each $i$, Gaussianity and the definition of $z_\delta$ give
\[
\Pr\left(|\xi_i|>z_\delta s_i\right)=\frac{\delta}{m}.
\]
A union bound shows that all $m$ inequalities hold simultaneously with probability at least $1-\delta$. On this event the true coefficient vector $\theta^\star$ belongs to $\Cset_\delta$. Hence, for every target $M$,
\[
L_\delta(M)\le\Risk_{\theta^\star}(M)\le U_\delta(M).
\]
Because the same event and the same feasible coefficient vector apply to all $M$, coverage is simultaneous over any number of targets. Sharpness relative to $\Cset_\delta$ follows directly from minimizing and maximizing the linear target functional over that set.
\end{proof}


\section{Proofs for Continuous-Exponent \ScaleCert}

\begin{proof}[Proof of Lemma~\ref{lem:secant}]
Fix a cell $[a,b]$, write $\Delta=b-a$, and let
$\lambda=(b-\alpha)/\Delta$. The interpolation error formula for a twice differentiable convex function gives
\begin{align*}
0 &\le \lambda\phi_a(M)+(1-\lambda)\phi_b(M)-\phi_\alpha(M)\\
  &\le \frac{\Delta^2}{8}
  \sup_{u\in[a,b]}\partial_u^2\phi_u(M).
\end{align*}
Termwise differentiation is justified by uniform convergence on every compact positive exponent interval and yields Eq.~\eqref{eq:basis-curvature}. This proves Eq.~\eqref{eq:secant-bound}.

For completeness, our implementation uses an explicit upper bound. Set
$g_a(x)=(\log x)^2x^{-1-a}$ and
$J=\max\{M,\lceil e^{2/(1+a)}\rceil\}$. The function $g_a$ decreases for $x\ge J$. Hence
\begin{align*}
\partial_a^2\phi_a(M)
&=\sum_{j=M+1}^{J}g_a(j)+\sum_{j>J}g_a(j)\\
&\le \sum_{j=M+1}^{J}g_a(j)+\int_J^\infty g_a(x)\,dx,
\end{align*}
where
\[
\int_J^\infty g_a(x)\,dx
=J^{-a}\left\{\frac{(\log J)^2}{a}
+\frac{2\log J}{a^2}+\frac{2}{a^3}\right\}.
\]
Because every term in Eq.~\eqref{eq:basis-curvature} decreases with the exponent, this also bounds the curvature over $[a,b]$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:continuous-certificate}]
Partition the true measure as $\mu=\sum_\ell\mu_\ell$, where $\mu_\ell$ is supported on
$[\gamma_\ell,\gamma_{\ell+1}]$. Define
\begin{align*}
p_\ell&=\int\frac{\gamma_{\ell+1}-\alpha}{\Delta_\ell}\,d\mu_\ell(\alpha),\\
q_\ell&=\int\frac{\alpha-\gamma_\ell}{\Delta_\ell}\,d\mu_\ell(\alpha).
\end{align*}
Both are nonnegative and $p_\ell+q_\ell=\mu_\ell([\gamma_\ell,\gamma_{\ell+1}])$. Integrating the pointwise secant inequalities from Lemma~\ref{lem:secant} gives, for every $M$,
\[
D_M\vartheta-G_M\vartheta
\le \Risk_{E,\mu}(M)\le D_M\vartheta.
\]
On the event $\ell_i\le\Risk_{E,\mu}(M_i)\le u_i$ for every pilot,
$D_{M_i}\vartheta\ge\ell_i$ and
$(D_{M_i}-G_{M_i})\vartheta\le u_i$, so the mapped vector lies in
$\widetilde\Cset_\delta$. The target inequalities and the definitions of the two LP objectives then imply Eq.~\eqref{eq:continuous-coverage}. Gaussianity plus a Bonferroni union bound gives the pilot-band event probability at least $1-\delta$, and the same event covers every target simultaneously.

Finally, for the vector mapped from the true measure,
\begin{align*}
G_M\vartheta
&=\sum_\ell\varepsilon_\ell(M)(p_\ell+q_\ell)\\
&\le W\max_\ell\varepsilon_\ell(M)
\end{align*}
when $\mu$ has total mass at most $W$. Thus the true curve and its secant representation differ by at most this amount. Since each $\varepsilon_\ell(M)$ is quadratic in $\Delta_\ell$, the stated discretization rate follows.
\end{proof}

\section{Proof of the Finite-Sample OLS Bridge}

\begin{proof}[Proof of Proposition~\ref{prop:ols}]
After whitening, write the first $M$ coordinates as $X\sim\Normal(0,I_M)$. The omitted coordinates and label noise form a residual $\eta_M$ independent of $X$ with variance
$\E\eta_M^2=\Risk(M)$. Gaussianity of this residual is not needed for the mean identity. For a training design matrix $\mathbf X\in\R^{n\times M}$ and residual vector $\boldsymbol\eta$, ordinary least squares satisfies
\[
\widehat\theta-\theta=(\mathbf X^\top\mathbf X)^{-1}\mathbf X^\top\boldsymbol\eta.
\]
On an independent test point, conditional test risk is
$\Risk(M)+\|\widehat\theta-\theta\|_2^2$. Conditional on $\mathbf X$,
\[
\E_{\boldsymbol\eta}\|\widehat\theta-\theta\|_2^2
=\Risk(M)\operatorname{tr}\{(\mathbf X^\top\mathbf X)^{-1}\}.
\]
Since $\mathbf X^\top\mathbf X$ is Wishart with $n$ degrees of freedom \citep{muirhead1982aspects},
$\E(\mathbf X^\top\mathbf X)^{-1}=I_M/(n-M-1)$ for $n>M+1$. Therefore
\begin{align*}
\E\widehat\Risk_{\mathrm{OLS}}(M)
&=\Risk(M)\left(1+\frac{M}{n-M-1}\right)\\
&=\Risk(M)\frac{n-1}{n-M-1}.
\end{align*}

For the distributional identity, additionally assume that $\eta_M$ is Gaussian. Conditional Gaussianity then gives
$(\widehat\theta-\theta)/\sqrt{\Risk(M)}$ as a multivariate Student variable obtained by mixing a standard normal vector with an inverse-Wishart matrix. The corresponding squared norm has the Hotelling form
\[
\frac{\|\widehat\theta-\theta\|_2^2}{\Risk(M)}
\overset d=\frac{M}{n-M+1}F_{M,n-M+1},
\]
which proves Eq.~\eqref{eq:ols-distribution}.
\end{proof}

\section{Additional Experimental Details}

\paragraph{Exact constructions.}
The zeta basis is evaluated with SciPy and all LPs use the HiGHS backend. The matched-pair and hidden-crossover settings are generated by \texttt{experiments/run\_all.py}. The latter also reproduces the compute-to-target ratios reported in the main text.

\paragraph{Continuous refinement.}
The off-grid curve has floor $0.18$, exponents $(0.337,0.873)$, weights $(0.004,0.12)$, ten integer pilot sizes geometrically spaced from $16$ to $4096$, target $10^6$, and standard error $2\times10^{-5}$. The exponent interval is $[0.15,1.2]$. We use 11, 21, 41, 81, and 161 equally spaced grid points.

\paragraph{Finite-sample OLS.}
The oracle curve has floor $0.02$, exponents $(0.35,0.90)$, and weights $(0.004,0.12)$. We use $n=8000$, dimensions
$\{16,24,32,48,64,96,128,192,256\}$, and 5,000 draws from Eq.~\eqref{eq:ols-distribution}. The certificate uses dimensions through $96$ and an exponent grid of 43 points over $[0.15,1.20]$. A separate direct Gaussian-design unit test verifies the analytic sampler.

\paragraph{Synthetic benchmark.}
Each family uses nine pilot scales geometrically spaced from $1$ to $128$, target $10^6$, independent standard error $10^{-3}$, and 30 repetitions. The families $(E;\alpha;w)$ are
$(0.20;(0.55);(0.15))$,
$(0.20;(0.10,0.80);(3\times10^{-4},0.15))$, and
$(0.18;(0.12,0.45,1.10);(2.5\times10^{-4},0.020,0.12))$.
Bootstrap intervals use 20 refits and nominal coverage $0.90$; Laplace intervals use 1,200 posterior-normal draws. The broad certificate spans $[0.05,1.5]$ with 31 grid points.

\paragraph{Public curves.}
We use the NSL-PFN data at repository revision
\texttt{10a5a609}; the runner verifies SHA-256 checksums before analysis. The benchmark comprises five NMT and four image-classification curves, each evaluated at pilot fractions $0.50$, $0.70$, and $0.85$. No independent standard errors are available. We therefore solve an LP for the minimum uniform in-range radius and report held-out sensitivity to radius multipliers $1$--$4$. This experiment is descriptive and is not used to validate Theorem~\ref{thm:continuous-certificate}.

\paragraph{Reproducibility.}
The submission pipeline runs every stage in a fresh process through
\texttt{experiments/run\_submission.py}. Machine-readable outputs include per-run predictions and aggregate tables. All experiments are CPU-only; CI uses Ubuntu and Python~3.11 with the package ranges in \texttt{pyproject.toml}. Random experiments use fixed numerical seeds, and public-curve seeds are derived from SHA-256 identifiers rather than process-dependent hashes. GitHub Actions executes the unit tests, exact-risk script, all submission stages, and the manuscript build.
